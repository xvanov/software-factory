1"""Gate: ``tests-meaningful``. 

2 

3Runs the slop detector against the PR's test diff. Zero findings → pass. 

4Any finding → fail with the findings as details. 

5 

6If the app opts into mutation testing (``app_config.gates.mutation_testing 

7== True``), this gate ALSO would invoke mutmut/stryker — but Phase 4 just 

8wires the hook (sets a placeholder flag in details) without shipping a 

9working mutmut config. Apps opt-in later via their config.yaml. 

10""" 

11 

12from __future__ import annotations 

13 

14from factory.app_config import AppConfig 

15from factory.chain.gates.evaluator import GateResult, PRContext 

16from factory.chain.slop_detector import scan_diff 

17 

18 

19def evaluate(pr: PRContext, app_config: AppConfig) -> GateResult: 

20 label = "tests-meaningful" 

21 findings = scan_diff(pr.files_changed, repo_root=pr.repo_root) 

22 findings_dicts = [fnd.as_dict() for fnd in findings] 

23 if findings: 

24 return GateResult( 

25 label=label, 

26 passed=False, 

27 reason=f"{len(findings)} slop finding(s)", 

28 details={"findings": findings_dicts}, 

29 ) 

30 

31 # Mutation hook (opt-in). Phase 4 only wires the flag — no mutmut 

32 # subprocess invocation yet because real-run requires app-specific 

33 # mutmut configuration the factory cannot synthesize. P5.0 MEDIUM-3 

34 # carry-over: when an app opts in but no runner is wired the gate 

35 # must FAIL rather than silently pass; otherwise mutation_testing=true 

36 # is a no-op that gives operators false confidence. 

37 if app_config.gates.mutation_testing: 

38 return GateResult( 

39 label=label, 

40 passed=False, 

41 reason="mutation_testing opted-in but no runner wired", 

42 details={"mutation_status": "opted_in_no_runner", "findings": []}, 

43 ) 

44 

45 return GateResult( 

46 label=label, 

47 passed=True, 

48 reason="no slop findings", 

49 details={"mutation_status": "skipped", "findings": []}, 

50 )
