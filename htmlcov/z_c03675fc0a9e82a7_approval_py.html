1"""Operator-approval gate for machine-filed directions. 

2 

3The treadmill this closes 

4------------------------- 

5A scheduled scanner persona (``ralph``, ``bug_hunter``, ``security``, 

6``ux_auditor``) can file its own work orders: every run it looks for 

7something to complain about, and it always finds something. On 2026-07-24 the 

8UX auditor filed D114/115/116 in ~2h (~$145 projected EOD spend); on 

92026-07-30 it filed 015/016/017 — every one of them asking for *better inputs 

10for the auditor itself* — and ``auto_pm_sync`` triaged them into stories 

11WITHOUT an operator ever seeing them. Four PRs (#165/#166/#167/#169) had to be 

12closed by hand. 

13 

14Throttling the schedule only slows the treadmill down; the structural fix is 

15that **a direction the factory filed for itself must not reach the build 

16pipeline until a human says yes.** This module is that predicate, and 

17``factory.chain.pm_sync`` (the single door from direction → stories) enforces 

18it. 

19 

20Fail-safe by construction 

21------------------------- 

22The rule is an ALLOWLIST of sources that may auto-build, not a denylist of 

23scanner personas: 

24 

25* human-filed (``operator*``, ``cli*``, ``user*``, ``human*``) → auto-builds, 

26 exactly as before. The human path is not slowed down at all. 

27* deterministic, externally-triggered detectors (``ci-health``, 

28 ``flake-quarantine``, ``github``/``github_issue`` intake of a *user-filed* 

29 issue) → auto-builds. These are not opinions: a required check went red, a 

30 test flaked, a person opened an issue. 

31* everything else — every ``scheduled-<persona>`` source, any future 

32 machine filer, **and any direction whose source cannot be determined** 

33 (missing/corrupt ``state.yaml``) → requires an explicit operator approval. 

34 

35Approval is recorded in the direction's ``state.yaml`` under 

36``operator_approval`` (the same file that already carries ``source``, 

37``status`` and the ``audit`` trail), so it is human-readable, human-editable 

38and survives a restart. ``factory approve-direction`` is the supported way to 

39write it; ``factory inbox`` lists everything still waiting. 

40""" 

41 

42from __future__ import annotations 

43 

44from datetime import UTC, datetime 

45from typing import Any

47from factory.directions.parser import Direction

49#: ``state.yaml`` key holding the approval record. 

50APPROVAL_KEY = "operator_approval" 

51 

52#: Source prefix used by every cron-scheduled persona (see 

53#: ``factory.chain.scheduled_tasks``). Directions carrying it are exactly the 

54#: self-filed work orders this gate exists to stop. 

55SCHEDULED_SOURCE_PREFIX = "scheduled-" 

56 

57#: Sources whose *family* is a human. Prefix-matched because operators tag 

58#: free-form suffixes in practice (``operator-loop3``, ``cli-tell``, …). 

59_HUMAN_SOURCE_PREFIXES: tuple[str, ...] = ("operator", "cli", "user", "human") 

60 

61#: Non-human sources that may still auto-build: deterministic detectors whose 

62#: trigger is an objective external fact, not an LLM's opinion about what the 

63#: factory should work on next. Exact match — a new machine filer must be 

64#: added here deliberately, and until it is, it needs operator approval. 

65_DETERMINISTIC_SOURCES: frozenset[str] = frozenset( 

66 { 

67 "ci-health", # a REQUIRED check went red on main (D004) 

68 "flake-quarantine", # a test was observed flaking 

69 "github", # webhook intake of a human-filed issue 

70 "github_issue", # `factory ingest-issue` of a human-filed issue 

71 } 

72) 

73 

74#: Statuses at which a direction is still waiting to enter the pipeline. Only 

75#: these can be "awaiting approval" — an already-triaged direction is past 

76#: this gate. 

77PENDING_STATUSES: frozenset[str] = frozenset({"created", "needs-direction"})

80def direction_source(direction: Direction) -> str: 

81 """Return the recorded ``source`` for ``direction``, or ``""`` when unknown. 

82 

83 Reads ``direction.state``, i.e. the ``state.yaml`` projection. That file is 

84 gitignored (D018), so callers that want the AUTHORITATIVE value must resolve 

85 it from the ``directions`` row first — see 

86 :func:`factory.directions.watcher.hydrate_direction_source`, which 

87 ``pending_directions`` (and therefore the pm-sync gate) applies before this 

88 predicate ever runs. 

89 

90 Reading the file alone was the whole of the outage: D012 gave the table no 

91 ``source`` column, D018 stopped tracking the file, and this gate's 

92 "unknown ⇒ park" fail-safe then parked EVERY direction as soon as an operator 

93 followed the documented ``directions-regenerate-state`` recovery path. 

94 

95 A genuinely unknown source still yields ``""`` — which 

96 :func:`requires_operator_approval` treats as "needs a human", never as 

97 "safe to build". 

98 """ 

99 state = direction.state or {} 

100 if not isinstance(state, dict): 

101 return "" 

102 raw = state.get("source") 

103 return str(raw).strip() if raw not in (None, "") else ""

106def is_scheduled_persona_source(source: str) -> bool: 

107 """True when ``source`` names a cron-scheduled persona (``scheduled-*``).""" 

108 return source.startswith(SCHEDULED_SOURCE_PREFIX)

111def requires_operator_approval(direction: Direction) -> bool: 

112 """True when ``direction`` may not be auto-triaged without an operator. 

113 

114 Allowlist semantics (see the module docstring): only a human-filed source 

115 or a known deterministic detector auto-builds. Anything else — including 

116 an unresolvable source — requires approval. 

117 """ 

118 source = direction_source(direction) 

119 if not source: 

120 # Fail SAFE: we cannot prove a human asked for this. 

121 return True 

122 if source.startswith(_HUMAN_SOURCE_PREFIXES): 

123 return False 

124 if source in _DETERMINISTIC_SOURCES: 

125 return False 

126 return True

129def approval_record(direction: Direction) -> dict[str, Any] | None: 

130 """Return the ``operator_approval`` block from ``state.yaml``, if valid.""" 

131 state = direction.state or {} 

132 if not isinstance(state, dict): 

133 return None 

134 record = state.get(APPROVAL_KEY) 

135 if isinstance(record, dict): 

136 return record 

137 return None

140def is_operator_approved(direction: Direction) -> bool: 

141 """True when an operator explicitly approved this direction. 

142 

143 Requires ``operator_approval.approved is True`` (exactly ``True``, not a 

144 truthy string) AND a non-empty ``approved_by``: an approval nobody signed 

145 is not an approval. 

146 """ 

147 record = approval_record(direction) 

148 if record is None: 

149 return False 

150 if record.get("approved") is not True: 

151 return False 

152 return bool(str(record.get("approved_by") or "").strip())

155def is_auto_buildable(direction: Direction) -> bool: 

156 """True when the chain may triage ``direction`` without a human. 

157 

158 The single predicate ``pm_sync`` gates on. 

159 """ 

160 if not requires_operator_approval(direction): 

161 return True 

162 return is_operator_approved(direction)

165def awaiting_operator_approval(direction: Direction) -> bool: 

166 """True when ``direction`` is parked at this gate and needs an operator. 

167 

168 Narrower than ``not is_auto_buildable``: only directions still at a 

169 pending status (``created`` / ``needs-direction``) are *waiting* — a 

170 closed or already-validated one is not in the operator's inbox. 

171 """ 

172 if direction.status not in PENDING_STATUSES: 

173 return False 

174 return not is_auto_buildable(direction)

177def approval_blocked_reason(direction: Direction) -> str: 

178 """Short, operator-facing reason this direction is parked at the gate.""" 

179 source = direction_source(direction) 

180 if not source: 

181 return "source unknown (fail-safe: treated as machine-filed)" 

182 if is_scheduled_persona_source(source): 

183 persona = source[len(SCHEDULED_SOURCE_PREFIX) :] or "?" 

184 return f"auto-filed by scheduled persona '{persona}'" 

185 return f"machine-filed (source={source})"

188def approve_direction( 

189 direction: Direction, 

190 *, 

191 by: str, 

192 note: str | None = None, 

193 now: datetime | None = None, 

194) -> dict[str, Any]: 

195 """Record an operator approval on ``direction`` and return the record. 

196 

197 Writes the ``operator_approval`` block plus an audit entry into 

198 ``state.yaml`` via ``watcher.merge_state`` (which also refreshes 

199 ``direction.state`` in memory). Raises ``ValueError`` when ``by`` is 

200 empty — an unsigned approval is not an approval. 

201 """ 

202 signer = (by or "").strip() 

203 if not signer: 

204 raise ValueError("approve_direction requires a non-empty 'by' (who approved)") 

205 

206 # Imported here (not at module import) to keep this module dependency-light 

207 # for the pure predicates above. 

208 from factory.directions.watcher import _read_state_yaml, merge_state 

209 

210 ts = (now or datetime.now(UTC)).isoformat() 

211 record: dict[str, Any] = { 

212 "approved": True, 

213 "approved_by": signer, 

214 "approved_at": ts, 

215 } 

216 if note: 

217 record["note"] = note 

218 

219 # Append to the audit trail as it is ON DISK, not as it was when this 

220 # Direction was parsed: ``merge_state`` merges into the current file, so 

221 # writing a stale list back would silently drop entries added in between. 

222 on_disk = _read_state_yaml(direction.dir_path / "state.yaml") 

223 audit = on_disk.get("audit") 

224 audit = list(audit) if isinstance(audit, list) else [] 

225 audit.append( 

226 { 

227 "ts": ts, 

228 "by": signer, 

229 "event": "operator_approved", 

230 "details": {"note": note} if note else {}, 

231 } 

232 ) 

233 merge_state(direction, {APPROVAL_KEY: record, "audit": audit}) 

234 return record
