1"""The gutted-implementation control (019 AC2): a catch-all HTTP stub. 

2 

3Pure stdlib, in-process, no app tree involved. This is what the acceptance 

4oracle runs against BEFORE any real boot happens: a server that answers every 

5request with a fixed ``200 {}`` — a no-op implementation of the story's whole 

6surface. A criterion that STILL passes here could be satisfied by a no-op and 

7must never be credited (see ``factory.chain.gates.acceptance_verified``); an 

8oracle that sends zero requests, or one where EVERY criterion passes the stub, 

9is ``vacuous_oracle`` and blocks before a single real boot is paid for. 

10 

11Deliberately a catch-all rather than a route-scoped 404 responder: a fixed 

12``200 {}`` for every path and every method is the STRICTER control — a 

13criterion that only checks a status code or an absence is satisfied by it, so 

14it is the control that catches the most vacuous criteria, which is the 

15direction this control must err in. (A route-scoped 404 would be MORE 

16permissive: a criterion asserting "the route returns 404" would pass a 404 

17stub even though a real broken route also returns 404, hiding the vacuity.) 

18 

19TWO VARIANTS, and crediting requires failing BOTH (found 2026-08-07). The 

20single ``{}`` (empty) variant is too PERMISSIVE in one specific, likely shape: 

21an LLM-authored criterion like ``assert r.status_code == 200; items = 

22r.json()["items"]; assert len(items) >= 0`` fails the empty stub with a 

23``KeyError`` on ``"items"`` — which LOOKS like discriminating evidence, but 

24the actual boolean the author wrote (``len(items) >= 0``) is true for ANY 

25list, including the empty one a real no-op would return. The failure is an 

26accident of the stub's THINNESS, not of the assertion's strength. The 

27``plausible`` variant answers with a body carrying commonly-shaped keys 

28(``items: []``, ``id``, ``data``, …) precisely so that a criterion whose only 

29real content is "the response has roughly this shape" PASSES it — correctly 

30outing the criterion as vacuous — while a criterion that checks a SPECIFIC 

31value (``items[0]["email"] == "..."``) still fails it (``IndexError``) and 

32stays credited. A criterion must fail BOTH variants to enter ``K``; passing 

33either one is enough to exclude it. Kept in-process and cheap (two catch-all 

34stdlib servers, no boot) rather than app-specific — this is a heuristic 

35widening of the control, not a claim that it enumerates every possible 

36plausible shape. 

37 

38Every request's body is drained before responding (Content-Length or 

39chunked): skipping this leaves unread bytes on the socket, ``httpx`` sees a 

40connection reset on keep-alive, and a criterion that legitimately would have 

41failed the stub instead reports a client-side ERROR that looks like 

42discriminating evidence. ``protocol_version = "HTTP/1.1"`` is what makes that 

43drain logic load-bearing rather than decorative: HTTP/1.0 (the stdlib 

44default) closes the connection after every response, so a client never 

45reuses one and never observes an undrained body corrupting the NEXT request 

46on the same connection — the exact bug the drain exists to prevent. 

47""" 

48

58#: Bumped whenever the stub's *behaviour* changes (its response shape, its 

59#: draining logic, which methods it answers) — the stub run is cached on 

60#: ``(oracle_sha, STUB_VERSION, RUNNER_VERSION)``, and a stale cache entry from 

61#: a behaviourally different stub must not be reused as if nothing changed. 

62#: Bumped 2026-08-07: the ``plausible`` variant + HTTP/1.1 keep-alive. 

63STUB_VERSION = 2 

64 

65_EMPTY_BODY = b"{}" 

66_PLAUSIBLE_BODY = json.dumps( 

67 { 

68 "id": "stub", "items": [], "results": [], "data": {}, "count": 0, 

69 "total": 0, "status": "ok", "message": "stub", "name": "stub", 

70 "email": "stub@example.com", "value": None, "ok": True, "success": True, 

71 } 

72).encode("utf-8") 

73 

74#: The variant names crediting must fail BOTH of. 

75STUB_VARIANTS = ("empty", "plausible") 

76_MAX_LOG = 200

79class _StubHandler(BaseHTTPRequestHandler): 

80 # Set by ``stub_app()`` on the per-instance handler subclass before the 

81 # server starts; a plain class attribute would leak counts across 

82 # concurrent stub instances in the same test process. 

83 protocol_version = "HTTP/1.1" 

84 _counter: list[int] 

85 _log: deque[str] 

86 _body: bytes 

87 

88 def _drain_body(self) -> None: 

89 length = self.headers.get("Content-Length") 

90 if length is not None: 

91 try: 

92 n = int(length) 

93 except ValueError: 

94 n = 0 

95 if n > 0: 

96 self.rfile.read(n) 

97 return 

98 if (self.headers.get("Transfer-Encoding") or "").strip().lower() == "chunked": 

99 # Best-effort chunked drain — good enough for an oracle's httpx 

100 # client, which does not send chunked request bodies by default. 

101 while True: 

102 try: 

103 size_line = self.rfile.readline(64).strip() 

104 size = int(size_line, 16) if size_line else 0 

105 except (ValueError, OSError): 

106 return 

107 if size <= 0: 

108 self.rfile.readline(2) 

109 return 

110 self.rfile.read(size) 

111 self.rfile.read(2) 

112 

113 def _respond(self) -> None: 

114 self._drain_body() 

115 self._counter[0] += 1 

116 self._log.append(f"{self.command} {self.path}") 

117 if self.command == "HEAD": 

118 self.send_response(200) 

119 self.send_header("Content-Type", "application/json") 

120 self.send_header("Content-Length", str(len(self._body))) 

121 self.end_headers() 

122 return 

123 self.send_response(200) 

124 self.send_header("Content-Type", "application/json") 

125 self.send_header("Content-Length", str(len(self._body))) 

126 self.end_headers() 

127 try: 

128 self.wfile.write(self._body) 

129 except (BrokenPipeError, ConnectionResetError): 

130 pass 

131

157class StubApp: 

158 """A live catch-all HTTP stub. ``request_count``/``requests`` update live.""" 

159 

160 def __init__(self, base_url: str, counter: list[int], log: deque[str]) -> None: 

161 self.base_url = base_url 

162 self._counter = counter 

163 self._log = log 

164 

165 @property 

166 def request_count(self) -> int: 

167 return self._counter[0] 

168 

169 @property 

170 def requests(self) -> list[str]: 

171 return list(self._log) 

172 

173 

174@contextmanager 

175def stub_app(variant: str = "empty") -> Iterator[StubApp]: 

176 """A catch-all HTTP stub on an ephemeral port, torn down on exit. 

177 

178 ``variant``: ``"empty"`` (fixed ``200 {}``) or ``"plausible"`` (a body 

179 with commonly-shaped keys — see the module docstring for why crediting 

180 needs both). In-process (a daemon thread), not a subprocess — there is 

181 nothing to boot and nothing that can leak a process group: the whole 

182 point of the stub is that it is the factory's OWN trusted code, never 

183 the diff's. 

184 """ 

185 if variant not in STUB_VARIANTS: 

186 raise ValueError(f"unknown stub variant {variant!r}, expected one of {STUB_VARIANTS!r}") 

187 body = _EMPTY_BODY if variant == "empty" else _PLAUSIBLE_BODY 

188 counter = [0] 

189 log: deque[str] = deque(maxlen=_MAX_LOG) 

190 

191 class _Bound(_StubHandler): 

192 pass 

193 

194 _Bound._counter = counter 

195 _Bound._log = log 

196 _Bound._body = body 

197 

198 server = ThreadingHTTPServer(("127.0.0.1", 0), _Bound) 

199 port = server.server_address[1] 

200 thread = threading.Thread(target=server.serve_forever, daemon=True, name="factory-stub") 

201 thread.start() 

202 try: 

203 yield StubApp(base_url=f"http://127.0.0.1:{port}", counter=counter, log=log) 

204 finally: 

205 server.shutdown() 

206 server.server_close() 

207 thread.join(timeout=5) 

208 

209 

210__all__ = ["STUB_VARIANTS", "STUB_VERSION", "StubApp", "stub_app"]
