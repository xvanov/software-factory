244 # routes a RED ``test_command`` through flake detection: any failing test is 

245 # re-run in isolation; a test that flaps (fails then passes) is QUARANTINED

297def load_app_config(app: str, software_factory_root: Path) -> AppConfig: 

298 """Load and validate ``apps/<app>/config.yaml`` from the factory root. 

299 

300 If ``app_repo_path`` is unset in the YAML, it defaults to ``../<name>`` 

301 relative to the factory root (e.g. factory at ``~/software-factory/`` 

302 and apps as sibling directories at ``~/<name>/``). 

303 """ 

304 cfg_path = Path(software_factory_root) / "apps" / app / "config.yaml" 

305 if not cfg_path.exists(): 

306 raise FileNotFoundError(f"App config missing: {cfg_path}. Expected apps/<app>/config.yaml.") 

307 raw = yaml.safe_load(cfg_path.read_text(encoding="utf-8")) or {} 

308 if not isinstance(raw, dict): 

309 raise ValueError(f"{cfg_path}: top-level must be a YAML mapping") 

310 if not raw.get("app_repo_path"): 

311 # Mirror the documented convention: app source lives at a sibling 

312 # of the factory root, named by the app. 

313 raw["app_repo_path"] = f"../{raw.get('name') or app}" 

314 return AppConfig.model_validate(raw)

317def resolve_app_repo_path(cfg: AppConfig, software_factory_root: Path) -> Path: 

318 """Resolve ``cfg.app_repo_path`` against the factory root. 

319 

320 Absolute paths are returned unchanged; relative paths are anchored at 

321 ``software_factory_root``. The result is NOT required to exist — 

322 callers handle the "no app tree yet" case (e.g. context loader emits 

323 the NO CONTEXT AVAILABLE notice). 

324 """ 

325 raw = (cfg.app_repo_path or "").strip() 

326 if not raw: 

327 raw = f"../{cfg.name}" 

328 p = Path(raw) 

329 if p.is_absolute(): 

330 return p 

331 return (Path(software_factory_root) / p).resolve()

334def list_apps(software_factory_root: Path) -> list[dict[str, object]]: 

335 """Discover every ``apps/*/config.yaml`` and return one summary dict per app. 

336 

337 Each dict contains at least: ``name``, ``repo``, ``self_tick_enabled``, 

338 ``deploy_enabled`` — reading effective values from the app's config.yaml. 

339 The function is pure: it never mutates config, filesystem, or runtime state. 

340 """ 

341 apps_dir = Path(software_factory_root) / "apps" 

342 if not apps_dir.is_dir(): 

343 return [] 

344 

345 result: list[dict[str, object]] = [] 

346 for cfg_path in sorted(apps_dir.glob("*/config.yaml")): 

347 app_name = cfg_path.parent.name 

348 cfg = load_app_config(app_name, software_factory_root) 

349 result.append( 

350 { 

351 "name": cfg.name, 

352 "repo": cfg.repo, 

353 "self_tick_enabled": cfg.self_tick_enabled, 

354 "deploy_enabled": cfg.deploy.enabled, 

355 } 

356 ) 

357 return result
