1"""Deterministic toolchain detection for ``factory app add``. 

2 

3Reads a checked-out repository and proposes the four command gates an app 

4config carries (``lint_command``, ``format_check_command``, 

5``type_check_command``, ``test_command``) plus an optional smoke command. 

6 

7Pure and read-only: it opens files, never runs anything, never writes. Every 

8proposal carries the file that justified it (``sources``), and a gate nothing 

9justifies is reported in ``missing`` and left EMPTY — never filled with a 

10guess. An empty gate is "not configured", not "passing"; see 

11``docs``/the ``factory app add`` output for what that means per gate. 

12 

13Detection order per gate (first hit wins): 

14 

151. ``Makefile`` targets — the commands the humans already run, so the 

16 factory's gates and local development cannot drift apart (the same reason 

17 ``apps/template-probe/config.yaml`` uses ``npm run`` scripts). 

182. The root toolchain(s): Python (uv / poetry / pip), Node/TS (npm / pnpm / 

19 yarn / bun scripts), Go, Rust, .NET, JVM (Maven / Gradle), Ruby (Bundler), 

20 PHP (Composer). 

213. Toolchains found one directory down (``backend/``, ``frontend/``, …), 

22 wrapped as ``(cd <dir> && <cmd>)``. When several subdirectories provide the 

23 same gate they are joined with ``&&`` so ALL must pass. 

24"""

137def _detect_python(d: Path) -> tuple[str, dict[str, str], list[str]] | None: 

138 pyproject = d / "pyproject.toml" 

139 reqs = sorted(d.glob("requirements*.txt")) 

140 if not (pyproject.is_file() or (d / "setup.py").is_file() or reqs): 

141 # No Python manifest, but a ruff config: the repo lints its Python 

142 # from here (e.g. a root ruff.toml over a backend/ package). 

143 for cfg in ("ruff.toml", ".ruff.toml"): 

144 if (d / cfg).is_file(): 

145 ruff_only = { 

146 "lint_command": "ruff check .", 

147 "format_check_command": "ruff format --check .", 

148 } 

149 return f"python ({cfg})", ruff_only, [] 

150 return None 

151 data = _load_toml(pyproject) if pyproject.is_file() else {} 

152 tool = data.get("tool", {}) if isinstance(data.get("tool"), dict) else {} 

153 # Every place a dependency name can be mentioned, as one searchable blob. 

154 blob = _read(pyproject).lower() + "".join(_read(r).lower() for r in reqs) 

155 blob += _read(d / "setup.cfg").lower() + _read(d / "setup.py").lower() 

156 

157 if (d / "uv.lock").is_file() or "uv" in tool: 

158 manager = "uv" 

159 extras = (data.get("project") or {}).get("optional-dependencies") or {} 

160 run = "uv run --extra dev" if "dev" in extras else "uv run" 

161 elif (d / "poetry.lock").is_file() or "poetry" in tool: 

162 manager, run = "poetry", "poetry run" 

163 else: 

164 manager, run = "pip", "python -m" 

165 

166 def mentions(name: str) -> bool: 

167 return re.search(rf"(^|[^a-z0-9_-]){re.escape(name)}([^a-z0-9_-]|$)", blob) is not None 

168 

169 gates: dict[str, str] = {} 

170 notes: list[str] = [] 

171 has_ruff = "ruff" in tool or (d / "ruff.toml").is_file() or (d / ".ruff.toml").is_file() 

172 has_ruff = has_ruff or mentions("ruff") 

173 if has_ruff: 

174 gates["lint_command"] = f"{run} ruff check ." 

175 gates["format_check_command"] = f"{run} ruff format --check ." 

176 else: 

177 if (d / ".flake8").is_file() or mentions("flake8"): 

178 gates["lint_command"] = f"{run} flake8 ." 

179 if "black" in tool or mentions("black"): 

180 gates["format_check_command"] = f"{run} black --check ." 

181 

182 mypy_cfg = tool.get("mypy") if isinstance(tool.get("mypy"), dict) else None 

183 if mypy_cfg is not None or (d / "mypy.ini").is_file() or mentions("mypy"): 

184 # With ``files`` configured mypy knows its own targets; otherwise ``.``. 

185 gates["type_check_command"] = f"{run} mypy" + ( 

186 "" if mypy_cfg and mypy_cfg.get("files") else " ." 

187 ) 

188 elif "pyright" in tool or (d / "pyrightconfig.json").is_file() or mentions("pyright"): 

189 gates["type_check_command"] = f"{run} pyright" 

190 

191 has_tests = (d / "tests").is_dir() or (d / "test").is_dir() or (d / "conftest.py").is_file() 

192 if mentions("pytest") or "pytest" in tool or has_tests: 

193 gates["test_command"] = f"{run} pytest -q" 

194 if manager == "pip": 

195 notes.append( 

196 "python (pip): commands use `python -m <tool>`; the gate's shell must " 

197 "already have the dev requirements installed" 

198 ) 

199 return f"python ({manager})", gates, notes

202def _node_pm(d: Path, pkg: dict[str, Any]) -> str: 

203 declared = str(pkg.get("packageManager") or "").split("@", 1)[0] 

204 if declared in ("npm", "pnpm", "yarn", "bun"): 

205 return declared 

206 if (d / "pnpm-lock.yaml").is_file(): 

207 return "pnpm" 

208 if (d / "yarn.lock").is_file(): 

209 return "yarn" 

210 if (d / "bun.lockb").is_file() or (d / "bun.lock").is_file(): 

211 return "bun" 

212 return "npm"

237def _detect_node(d: Path) -> tuple[str, dict[str, str], list[str]] | None: 

238 if not (d / "package.json").is_file(): 

239 return None 

240 pkg = _load_json(d / "package.json") 

241 pm = _node_pm(d, pkg) 

242 scripts = pkg.get("scripts") or {} 

243 scripts = scripts if isinstance(scripts, dict) else {} 

244 deps: set[str] = set() 

245 for key in ("dependencies", "devDependencies"): 

246 val = pkg.get(key) 

247 if isinstance(val, dict): 

248 deps.update(val) 

249 run = {"npm": "npm run", "pnpm": "pnpm run", "yarn": "yarn run", "bun": "bun run"}[pm] 

250 exe = {"npm": "npx", "pnpm": "pnpm exec", "yarn": "yarn run", "bun": "bunx"}[pm] 

251 

252 gates: dict[str, str] = {} 

253 for gate, names in _NODE_SCRIPTS.items(): 

254 for name in names: 

255 body = scripts.get(name) 

256 if not isinstance(body, str) or not body.strip(): 

257 continue 

258 if name == "test" and body.strip() == _NPM_PLACEHOLDER_TEST: 

259 continue 

260 gates[gate] = f"{run} {name}" 

261 break 

262 # Fallbacks on the tools themselves when no script names them. 

263 if "type_check_command" not in gates and (d / "tsconfig.json").is_file(): 

264 gates["type_check_command"] = f"{exe} tsc --noEmit" 

265 has_prettier = "prettier" in deps or any(d.glob(".prettierrc*")) or "prettier" in pkg 

266 if "format_check_command" not in gates and has_prettier: 

267 gates["format_check_command"] = f"{exe} prettier --check ." 

268 lang = "typescript" if (d / "tsconfig.json").is_file() or "typescript" in deps else "node" 

269 notes = [ 

270 f"{lang} ({pm}): gates run in a fresh git worktree with no node_modules — " 

271 "prefix an install step (e.g. `" 

272 + { 

273 "npm": "npm ci", 

274 "pnpm": "pnpm install --frozen-lockfile", 

275 "yarn": "yarn install --frozen-lockfile", 

276 "bun": "bun install --frozen-lockfile", 

277 }[pm] 

278 + " && …`) if the commands need dependencies" 

279 ] 

280 return f"{lang} ({pm})", gates, notes

283def _detect_jvm(d: Path) -> tuple[str, dict[str, str], list[str]] | None: 

284 gradle = [n for n in ("build.gradle", "build.gradle.kts") if (d / n).is_file()] 

285 if gradle: 

286 tool = "./gradlew" if (d / "gradlew").is_file() else "gradle" 

287 build = _read(d / gradle[0]).lower() 

288 gates = { 

289 "type_check_command": f"{tool} classes testClasses", 

290 "test_command": f"{tool} test", 

291 } 

292 if "spotless" in build: 

293 gates["format_check_command"] = f"{tool} spotlessCheck" 

294 if "checkstyle" in build: 

295 gates["lint_command"] = f"{tool} checkstyleMain" 

296 elif "ktlint" in build: 

297 gates["lint_command"] = f"{tool} ktlintCheck" 

298 return "jvm (gradle)", gates, [] 

299 if (d / "pom.xml").is_file(): 

300 tool = "./mvnw" if (d / "mvnw").is_file() else "mvn" 

301 pom = _read(d / "pom.xml").lower() 

302 gates = { 

303 "type_check_command": f"{tool} -B -q test-compile", 

304 "test_command": f"{tool} -B -q test", 

305 } 

306 if "spotless" in pom: 

307 gates["format_check_command"] = f"{tool} -B -q spotless:check" 

308 if "maven-checkstyle-plugin" in pom: 

309 gates["lint_command"] = f"{tool} -B -q checkstyle:check" 

310 return "jvm (maven)", gates, [] 

311 return None

314def _detect_ruby(d: Path) -> tuple[str, dict[str, str], list[str]] | None: 

315 if not (d / "Gemfile").is_file(): 

316 return None 

317 gemfile = _read(d / "Gemfile") + _read(d / "Gemfile.lock") 

318 gates: dict[str, str] = {} 

319 if "rubocop" in gemfile or (d / ".rubocop.yml").is_file(): 

320 gates["lint_command"] = "bundle exec rubocop" 

321 if "sorbet" in gemfile: 

322 gates["type_check_command"] = "bundle exec srb tc" 

323 elif "steep" in gemfile: 

324 gates["type_check_command"] = "bundle exec steep check" 

325 if "rspec" in gemfile or (d / "spec").is_dir(): 

326 gates["test_command"] = "bundle exec rspec" 

327 elif (d / "Rakefile").is_file() and (d / "test").is_dir(): 

328 gates["test_command"] = "bundle exec rake test" 

329 notes = ["ruby (bundler): gates run in a fresh worktree — prefix `bundle install && …`"] 

330 return "ruby (bundler)", gates, notes

333def _detect_php(d: Path) -> tuple[str, dict[str, str], list[str]] | None: 

334 if not (d / "composer.json").is_file(): 

335 return None 

336 pkg = _load_json(d / "composer.json")

343 scripts: dict[str, Any] = raw_scripts if isinstance(raw_scripts, dict) else {} 

344 gates: dict[str, str] = {} 

345 if "test" in scripts: 

346 gates["test_command"] = "composer run-script test" 

347 elif "phpunit/phpunit" in deps: 

348 gates["test_command"] = "vendor/bin/phpunit" 

349 elif "pestphp/pest" in deps: 

350 gates["test_command"] = "vendor/bin/pest" 

351 if "phpstan/phpstan" in deps: 

352 gates["type_check_command"] = "vendor/bin/phpstan analyse" 

353 elif "vimeo/psalm" in deps: 

354 gates["type_check_command"] = "vendor/bin/psalm" 

355 if "friendsofphp/php-cs-fixer" in deps: 

356 gates["format_check_command"] = "vendor/bin/php-cs-fixer fix --dry-run --diff" 

357 elif "laravel/pint" in deps: 

358 gates["format_check_command"] = "vendor/bin/pint --test" 

359 if "squizlabs/php_codesniffer" in deps: 

360 gates["lint_command"] = "vendor/bin/phpcs" 

361 notes = ["php (composer): gates run in a fresh worktree — prefix `composer install && …`"]

393def _detect_dotnet(d: Path) -> tuple[str, dict[str, str], list[str]] | None: 

394 slns = sorted([*d.glob("*.sln"), *d.glob("*.slnx")]) 

395 projs = sorted([*d.glob("*.csproj"), *d.glob("*.fsproj"), *d.glob("*.vbproj")]) 

396 if not slns and not projs: 

397 return None 

398 notes: list[str] = [] 

399 # ``dotnet`` refuses to guess when a folder holds more than one solution or 

400 # project file, so name the target explicitly whenever it is unambiguous. 

401 if len(slns) == 1: 

402 target = f" {slns[0].name}" 

403 elif not slns and len(projs) == 1: 

404 target = f" {projs[0].name}" 

405 else: 

406 target = "" 

407 if len(slns) + len(projs) > 1: 

408 notes.append( 

409 ".NET: several solution/project files at the root (" 

410 + ", ".join(p.name for p in [*slns, *projs]) 

411 + ") — the commands name none; add the right one by hand" 

412 ) 

413 tools = _load_json(d / ".config" / "dotnet-tools.json").get("tools") or {} 

414 restore = "dotnet tool restore && " if tools else "" 

415 if "csharpier" in tools: 

416 fmt = f"{restore}dotnet csharpier --check ." 

417 else: 

418 fmt = f"dotnet format{target} whitespace --verify-no-changes" 

419 gates = { 

420 # Analyzer diagnostics at warning+ severity, without rewriting anything. 

421 "lint_command": f"dotnet format{target} analyzers --verify-no-changes --severity warn", 

422 "format_check_command": fmt, 

423 # Compilation IS the type check for C#/F#. 

424 "type_check_command": f"dotnet build{target} --nologo", 

425 "test_command": f"dotnet test{target} --nologo", 

426 } 

427 return ".net", gates, notes

456def detect(repo: Path) -> Detection: 

457 """Propose gate commands for the repository checked out at ``repo``.""" 

458 repo = Path(repo) 

459 out = Detection() 

460 

461 # 1. Makefile targets. 

462 targets = makefile_targets(repo) 

463 for gate, names in _MAKE_CANDIDATES.items(): 

464 for name in names: 

465 if name in targets: 

466 out.gates[gate] = f"make {name}" 

467 out.sources[gate] = f"Makefile target `{name}`" 

468 break 

469 if targets: 

470 out.toolchains.append("make") 

471 if out.gates: 

472 out.notes.append( 

473 "Makefile targets are used as-is: check they do not assume a pre-built local " 

474 "env (.venv, node_modules) — gates run in a fresh git worktree" 

475 ) 

476 if "smoke" in targets: 

477 out.smoke_command = "make smoke" 

478 elif (repo / "scripts" / "smoke.sh").is_file(): 

479 out.smoke_command = "bash scripts/smoke.sh" 

480 

481 # 2 + 3. Root toolchains, then one directory down. Per gate and per 

482 # toolchain family, the SHALLOWEST provider wins: a root ``ruff check .`` 

483 # already covers ``backend/``, but says nothing about ``frontend/``'s npm 

484 # lint — so a gate collects one command per family, all joined with ``&&``. 

485 make_gates = set(out.gates) 

486 collected: dict[str, list[tuple[str, str, str]]] = {} # gate -> (family, where, cmd) 

487 

488 def _take(where: str, label: str, gates: dict[str, str]) -> None: 

489 fam = _family(label) 

490 for gate, cmd in gates.items(): 

491 if gate in make_gates: 

492 continue 

493 have = collected.setdefault(gate, []) 

494 if where and any(f == fam and not w for f, w, _ in have): 

495 continue # the root already covers this family 

496 have.append((fam, where, cmd)) 

497 

498 for label, gates, notes in _detect_dir(repo): 

499 out.toolchains.append(label) 

500 out.notes.extend(notes) 

501 _take("", label, gates) 

502 

503 subs = ( 

504 sorted( 

505 p 

506 for p in repo.iterdir() 

507 if p.is_dir() and p.name not in _SKIP_DIRS and not p.name.startswith(".") 

508 ) 

509 if repo.is_dir() 

510 else [] 

511 ) 

512 for sub in subs: 

513 for label, gates, notes in _detect_dir(sub): 

514 out.toolchains.append(f"{label} in {sub.name}/") 

515 out.notes.extend(f"{sub.name}/: {n}" for n in notes) 

516 _take(sub.name, label, gates) 

517 

518 for gate, entries in collected.items(): 

519 out.gates[gate] = " && ".join( 

520 cmd if not where else f"(cd {where} && {cmd})" for _, where, cmd in entries 

521 ) 

522 out.sources[gate] = ", ".join( 

523 f"{fam} at root" if not where else f"{fam} in {where}/" for fam, where, _ in entries 

524 ) 

525 

526 if not out.toolchains: 

527 out.notes.append( 

528 "no supported toolchain recognised (make, python, node/ts, go, rust, .net, " 

529 "jvm, ruby, php): every gate is EMPTY — fill them in by hand before " 

530 "submitting jobs, or the test gate can never go green" 

531 ) 

532 

533 # A template-derived repo bootstraps its own deps; every gate needs it in a 

534 # fresh worktree (see apps/template-probe/config.yaml). 

535 if (repo / "scripts" / "ensure-deps.sh").is_file(): 

536 for gate in list(out.gates): 

537 out.gates[gate] = f"bash scripts/ensure-deps.sh && {out.gates[gate]}" 

538 out.notes.append("scripts/ensure-deps.sh found: every gate is prefixed with it") 

539 

540 return out
