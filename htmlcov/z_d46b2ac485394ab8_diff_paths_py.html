1"""ONE path classifier for "is this file part of the oracle, or the fix?". 

2 

3Two subsystems have to answer that question and they must never disagree: 

4 

5* ``bench/swebench_adapter.py`` strips test edits out of a prediction before 

6 the hidden suite grades it, and REFUSES a prediction that edits a pytest 

7 collection channel; 

8* the ``production-tree-changed`` merge gate (``factory/chain/gates/``) blocks 

9 a story whose diff contains no production-code change at all. 

10 

11The bench harness owned these regexes first and they were measured against 

12real pytest behaviour (see the comments below). They live here so the gate 

13reuses the SAME definition instead of growing a second, drifting one — a 

14divergence would mean the benchmark and the chain disagree about what a 

15"production change" is, and the number the benchmark reports would stop 

16describing the thing the chain gates on. 

17 

18Definitions 

19----------- 

20* **test path** (``is_test_path``) — anything that might belong to a repo's test 

21 surface: test code, a ``testing/`` directory, ``conftest.py``, 

22 ``test_<anything>`` whatever its extension. Deliberately OVER-inclusive. 

23* **test CODE path** (``is_test_code_path``) — a file that unambiguously IS a 

24 test: under ``tests/``/``test/``, ``test_x.<code-ext>``, 

25 ``x_test.<code-ext>``, ``conftest.py``, ``*.spec.tsx``. A strict SUBSET of 

26 ``is_test_path`` — asserted by ``tests/test_diff_paths.py``, over this repo's 

27 own file list as well as fixed cases. 

28* **collection channel** — a file that decides WHICH tests run, or WHETHER 

29 they run, without being test code. 

30* **production path** — not test CODE, and not a collection channel. Docs, 

31 non-collection config and source all count: the predicate answers "did the 

32 change touch something other than the oracle", not "is this file 

33 interesting". 

34 

35Two matchers, one definition, opposite safe directions 

36------------------------------------------------------ 

37The bench STRIPS on its matcher, so over-inclusion is safe (stripping one file 

38too many can only weaken the arm's own patch) and under-inclusion would let an 

39arm edit the oracle judging it. The merge gate BLOCKS on its matcher, so 

40over-inclusion costs real merges. They therefore need the same notion pointed in 

41opposite directions, and the strict one is a subset of the broad one. 

42 

43This is not a theoretical split. MEASURED in this repo: the broad matcher calls 

44``factory/testing/flake.py`` — live merge-gate code, imported by 

45``gates/tests_green.py`` — and ``factory/personas/test_designer.md`` test files. 

46Used as a REQUIRED merge gate it would have refused any story whose diff is 

47confined to those; 2 of the last 400 commits on ``main`` are exactly that shape 

48(a ``factory_improver`` prompt_edit, and a persona clarification). Both matchers 

49live here and the subset relation is a test, so they cannot drift into 

50disagreeing about a real file. 

51 

52There is a THIRD, narrower test-path matcher in ``factory/chain/branch.py`` 

53(``is_test_file`` / ``_TEST_FILE_PATTERNS``). It is deliberately left alone: 

54it drives the legacy "dev must not modify frozen tests" diff check, and 

55widening it would change unrelated enforcement. Do not add a fourth. 

56"""

60import re

62# A tests?/ or test/ directory anywhere, test_foo.py, foo_test.go, conftest.py 

63# (fixtures shape the oracle too), *.spec.tsx, a testing/ directory. 

64_TEST_PATH = re.compile( 

65 r"(^|/)tests?(/|$)" # a tests/ or test/ directory anywhere 

66 r"|(^|/)test_[^/]+$" # test_foo.py 

67 r"|_test\.[a-z]+$" # foo_test.go 

68 r"|(^|/)conftest\.py$" # pytest fixtures shape the oracle too 

69 r"|\.spec\.[jt]sx?$" 

70 r"|(^|/)testing(/|$)" 

71) 

72 

73# Extensions a TEST is actually written in. A ``test_*``/``*_test`` file with any 

74# other extension (``.md``, ``.txt``, ``.json``, ``.csv``) is a document or a 

75# fixture, not a test — and the strict matcher is used where a false positive 

76# costs a merge, so it must not guess. 

77_TEST_CODE_EXT = ( 

78 r"(py|pyi|pyx|rb|go|js|jsx|mjs|cjs|ts|tsx|java|kt|kts|rs|c|cc|cpp|cxx|h|hpp" 

79 r"|cs|php|sh|bash|lua|pl|swift|scala|ex|exs|m|mm|dart|zig)" 

80) 

81 

82# The STRICT matcher — a strict subset of ``_TEST_PATH``. Drops the two branches 

83# that make the broad matcher over-inclusive: 

84# * ``(^|/)testing(/|$)``: a ``testing/`` package is production code in plenty 

85# of repos, including this one (``factory/testing/flake.py`` is imported by 

86# the tests-green merge gate); 

87# * an extension-agnostic ``test_[^/]+$``: it swallows 

88# ``factory/personas/test_designer.md``, a persona PROMPT — i.e. product. 

89_TEST_CODE_PATH = re.compile( 

90 r"(^|/)tests?(/|$)" 

91 rf"|(^|/)test_[^/]*\.{_TEST_CODE_EXT}$" 

92 rf"|_test\.{_TEST_CODE_EXT}$" 

93 r"|(^|/)conftest\.py$" 

94 r"|\.spec\.[jt]sx?$" 

95) 

96 

97# Files that decide WHICH tests run and WHETHER they run, without being test 

98# files themselves. ``_TEST_PATH`` matches test code; these are the collection 

99# and auto-import channels that reach the same outcome sideways: 

100# 

101# * ``pyproject.toml`` / ``setup.cfg`` / ``tox.ini`` / ``pytest.ini`` — 

102# ``[tool.pytest.ini_options] addopts``, ``-p <plugin>``, markers, 

103# ``collect_ignore``; 

104# * ``setup.py`` / ``noxfile.py`` — run at build/test time; 

105# * ``sitecustomize.py`` / ``usercustomize.py`` / ``*.pth`` — imported by 

106# the interpreter itself, before pytest starts; 

107# * ``*pytest*plugin*.py`` — an entry-point plugin. 

108# 

109# Measured against real pytest: a root ``_fixup.py`` with 

110# ``pytest_collection_modifyitems`` marking every item skipped plus 

111# ``addopts = "-p _fixup"`` makes ``python -m pytest <ids>`` exit 0 with 

112# "2 skipped". Under exit-code grading that was RESOLVED. The plugin file 

113# itself is ordinary production-looking code, so the CONFIG edit is the 

114# chokepoint worth refusing on. 

115# 

116# THREE MORE, each a forced pass measured 2026-08-05 against the acceptance 

117# oracle gate — every one of them classified PRODUCTION before this, so nothing 

118# rolled it back: 

119# 

120# * ``.pytest.ini`` — pytest honours the DOT-prefixed name as an inifile and it 

121# outranks ``pyproject.toml``. Every other name here is anchored on 

122# ``(^|/)``, which the leading dot walked straight past. Same attack as 

123# ``addopts = "-p _fixup"``, routed around the ``pyproject.toml`` rollback. 

124# * ``*.dist-info/`` and ``*.egg-info/`` — ``importlib.metadata`` scans every 

125# ``sys.path`` entry for these, and pytest calls 

126# ``load_setuptools_entrypoints("pytest11")``. An ``entry_points.txt`` 

127# committed in the run cwd auto-loads a dev-authored plugin with NO config 

128# edit at all. They are build artifacts; a diff has no business in them. 

129# * a module that SHADOWS THE RUNNER in the run cwd (``pytest.py``, 

130# ``_pytest.py``, ``py.py``, ``pluggy.py``). ``python -m pytest`` puts the cwd 

131# at ``sys.path[0]``, so ``pytest.py`` there *becomes* the whole "pytest run" 

132# — print a summary, ``sys.exit(0)``, done. 

133_COLLECTION_CHANNEL = re.compile( 

134 r"(^|/)(pyproject\.toml|setup\.cfg|tox\.ini|pytest\.ini|\.pytest\.ini|setup\.py" 

135 r"|noxfile\.py|sitecustomize\.py|usercustomize\.py|conftest\.py)$" 

136 r"|\.pth$" 

137 # The test runner's own import names. Shadowing any of them replaces the 

138 # thing that reports the verdict. 

139 r"|(^|/)(pytest|_pytest|py|pluggy)\.py$" 

140 # Installed-distribution metadata: the pytest11 entry-point channel. 

141 r"|(^|/)[^/]*\.(dist-info|egg-info)(/|$)" 

142 r"|(^|/)[^/]*pytest[^/]*plugin[^/]*\.py$" 

143 r"|(^|/)[^/]*plugin[^/]*pytest[^/]*\.py$" 

144)

147def _normalize(path: str) -> str: 

148 """``a/b`` form, no leading ``./``, backslashes folded to ``/``.""" 

149 norm = path.strip().replace("\\", "/") 

150 while norm.startswith("./"): 

151 norm = norm[2:] 

152 return norm.lstrip("/")

155def is_test_path(path: str) -> bool: 

156 """BROAD: anything that might belong to the test surface. Bench-side. 

157 

158 Over-inclusive on purpose — the bench strips these out of a prediction, and 

159 stripping one file too many can only weaken the arm's own patch. 

160 """ 

161 return bool(_TEST_PATH.search(_normalize(path))) 

162 

163 

164def is_test_code_path(path: str) -> bool: 

165 """STRICT: the file unambiguously IS a test. Gate-side. 

166 

167 A subset of ``is_test_path``. Used where a false positive BLOCKS a merge, so 

168 it never guesses: a ``test_*`` file whose extension is not a programming 

169 language is a document or a fixture, and a ``testing/`` package is ordinary 

170 production code. 

171 """ 

172 return bool(_TEST_CODE_PATH.search(_normalize(path))) 

173 

174 

175def is_collection_channel_path(path: str) -> bool: 

176 """True for a file that can change which tests run, or whether they run.""" 

177 return bool(_COLLECTION_CHANNEL.search(_normalize(path))) 

178 

179 

180def is_production_path(path: str) -> bool: 

181 """True when changing ``path`` changes the code under test. 

182 

183 Neither test CODE (the strict matcher — this predicate gates merges, and a 

184 false positive costs a real one) nor a collection channel. A story whose 

185 whole diff is test files plus ``pyproject.toml`` changed no production code, 

186 however many lines it moved. 

187 """ 

188 norm = _normalize(path) 

189 if not norm: 

190 return False 

191 return not is_test_code_path(norm) and not is_collection_channel_path(norm) 

192

194def production_paths(paths: list[str]) -> list[str]: 

195 """The subset of ``paths`` that is production code, order preserved.""" 

196 return [p for p in paths if is_production_path(p)]

199def _diff_target_paths(patch: str) -> list[str]: 

200 """Extract the list of file paths a unified diff touches. 

201 

202 Strips the ``a/`` / ``b/`` prefixes ``git diff`` emits, and dedupes 

203 while preserving order so callers can present a stable list to the 

204 operator. 

205 

206 Moved from ``factory/chain/factory_improver_apply.py`` (deleted 

207 2026-08-07 along with the L2/L4 self-improvement tiers) — this is the one 

208 remaining place that turns a raw unified diff into a path list, consumed 

209 by ``factory.manager.staging`` and the ``auto_merge`` self-edit guard. 

210 """ 

211 paths: list[str] = [] 

212 seen: set[str] = set() 

213 

214 def _add(raw: str) -> None: 

215 p = raw 

216 if p.startswith("a/") or p.startswith("b/"): 

217 p = p[2:] 

218 if p and p != "/dev/null" and p not in seen: 

219 seen.add(p) 

220 paths.append(p) 

221 

222 for line in patch.splitlines(): 

223 if line.startswith("diff --git "): 

224 # ``diff --git a/path/to/file b/path/to/file`` 

225 # Extract BOTH the a/ (source) AND b/ (destination) sides. A pure 

226 # 100%-similarity rename carries NO ``+++`` hunk header, so the 

227 # destination path lives ONLY on this line — e.g. a rename INTO 

228 # factory/ (``diff --git a/apps/x.py b/factory/evil.py``) would 

229 # otherwise be seen only as ``apps/x.py`` and evade the self-edit / 

230 # forbidden-path detection that both the staging gate and the 

231 # forbidden guard rely on. 

232 parts = line.split() 

233 if len(parts) >= 4: 

234 _add(parts[2]) # a/ (source) 

235 _add(parts[3]) # b/ (destination — the rename/copy target) 

236 elif line.startswith("+++ ") and not line.startswith("+++ /dev/null"): 

237 _add(line[4:].strip()) 

238 return paths
